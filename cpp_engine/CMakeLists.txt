cmake_minimum_required(VERSION 3.16)
project(surgical_telemetry_engine CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(include)
add_executable(engine src/main.cpp src/protocol_adapter.cpp)
target_link_libraries(engine pthread)
add_executable(test_engine tests/test_engine.cpp src/protocol_adapter.cpp)
target_link_libraries(test_engine pthread)
enable_testing()
add_test(NAME test_engine COMMAND test_engine)
